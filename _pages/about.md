Hi there! I am Yixin Liu, an ARC DECRA Fellow and Lecturer (Tenure-Track Assistant Professor equivalent in the US system) in the School of Information and Communication Technology (ICT), Griffith University, Australia. Before taking up this role, I worked as a Research Fellow at Griffith University starting in 2024. I am a core member of the TrustAGI Lab led by [Prof. Shirui Pan](https://shiruipan.github.io/). I received my Ph.D. from the Faculty of Information Technology, Monash University, in 2024, and obtained my B.S. and M.E. degrees from Beihang University, China, in 2017 and 2020, respectively. I was awarded the [Google PhD Fellowship in 2022](https://research.google/programs-and-events/phd-fellowship/recipients/) as one of five recipients from Australia.

  function initPubFilter() {

    var list = document.getElementById('pub-list');

    var buttons = bar.querySelectorAll('button');
    var cards = list.querySelectorAll('.pub-card');

      var btn = e.target.closest('button');

      if (!btn) return;

      buttons.forEach(function (button) {
        button.classList.remove('active');
      });

      btn.classList.add('active');

      cards.forEach(function (card) {
        var show =
          topic === 'all' ||
          card.getAttribute('data-topic') === topic;

        card.style.display = show ? '' : 'none';
      });
    });
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initPubFilter);
  } else {
    initPubFilter();
  }
})();
